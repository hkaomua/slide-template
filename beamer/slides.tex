\begin{frame}{数式の挿入}
  測定値を $x_i$、測定回数を $n$ とします。
  % 文中の数式は$...$、独立した数式は\[...\]で囲みます。
  \[
    \bar{x} = \frac{1}{n}\sum_{i=1}^{n} x_i
  \]
  記入例：$x_i = 10, 12, 14, 16, 18$、$n = 5$
  \[
    \bar{x} = \frac{10 + 12 + 14 + 16 + 18}{5} = 14
  \]
\end{frame}
